void Find_Zebra(uint8 *data)
{
    uint8 i,j,a=0;
    for(i=25;i<=60;i++)
    {
        for(j=56;j<64;j++)
        {
            if(data[i*COL+j])
            {
                a++;
            }
        }
        if(agl_blob_count(data, i, 1, COL-2) >= 2) a = 0;   // [2026-10-20] glare row
        if(a>2)
        {
            gpio_set_level(P33_10,1);
            Flag_stop=1;
            Rhombus_Flag = 1;
            break;
        }
    }
}

/* ---------------- Find_Decelerate（原第 2186~2200 行）---------------- */
void Point_Repair_Line(uint8 Start, uint8 Stop,uint16 position_start, uint16 position_stop ,uint16 *Line_Add)
{
    float res;
    uint8 i;    // 鎺у埗琛�
    float ka, kb;
    ka = 1.0*(position_start - position_stop) / (Start - Stop);
    kb = 1.0*position_start - (ka * Start);

    for (i = Stop; i <= Start; )
    {
        i += 1;
        res = i * ka + kb;
        Line_Add[i] = range_protect((int32)res, 2, (COL-2));
    }
}

/* ---------------- Point_Repair_Line（原第 2302~2358 行）---------------- */
void Turn_Left(void)
{
    if(Flag_turn == 0&&Flag_stop==0)
    {
        Flag_stop = 7;
        Flag_turn = 3;
        Distance_Test_ON = 1;
    }
    if(Flag_stop == 7&&Distance_Test>50)
    {
        Distance_Test = 0;
        Distance_Test_ON = 0;

        Light_Flag=2;

        Flag_turn = 1;
        Flag_stop = 2;

        Distance_ON = 1;
    }
    if(Flag_turn == 1 && Distance > 165)//180
    {
        Distance_ON = 0;
        Distance = 0;
        Light_Flag=0;

        Flag_turn = 6;//鍙宠浆鐭

        Flag_stop = 3;//鍙宠浆鐭閫熷害
        Distance_Test_ON = 1;
    }

    if(Flag_turn == 6 && Distance_Test > 85 )//鍙宠浆澶嶄綅
    {
        gpio_set_level(P33_10,1);//铚傞福鍣ㄥ叧闂�
        Distance_Test = 0;
        Distance_Test_ON = 0;

        Flag_turn = 0;
        Flag_stop = 5;
        Distance_ON = 1;
    }
    if(Flag_turn == 0 && Distance > 450 )//缁欎竴娈佃窛绂诲惊杩瑰浣�
    {
        gpio_set_level(P33_10,0);//铚傞福鍣ㄥ叧闂�
        Distance_ON = 0;
        Distance = 0;

        Flag_turn = 0;
        Flag_stop=0;

        Rhombus_Flag=0;
        Deal_Flag=2;
        Distance_Test_ON = 1;
    }

}
